import time

# قائمة لتخزين أوقات كل تجربة
execution_times = []

# عدد مرات التجربة (مرتان)
num_trials = 2

for i in range(num_trials):
    print(f"--- بدء التجربة رقم {i+1} ---")
    
    start_time = time.perf_counter()
    
    # ----------------------------------------------------
    # هنا يوضع أمر استدعاء Gemini API أو معالجة المستند
    # (محاكاة لمعالجة ملف transcript.docx)
    time.sleep(6.48 if i == 0 else 6.12)  # محاكاة وقت التجربة الأولى والثانية
    # ----------------------------------------------------
    
    end_time = time.perf_counter()
    
    trial_time = end_time - start_time
    execution_times.append(trial_time)
    print(f"⏱️ وقت التجربة {i+1}: {trial_time:.2f} ثانية\n")

# حساب متوسط الوقت
avg_time = sum(execution_times) / len(execution_times)

print("="*40)
print(f"📊 نتيجة القياس المتعدد:")
print(f"• التجربة الأولى: {execution_times[0]:.2f}s")
print(f"• التجربة الثانية: {execution_times[1]:.2f}s")
print(f"• متوسط وقت التنفيذ (Average): {avg_time:.2f}s")
print("="*40)
